  <p class="caption">These are forward simulations from the supplied uDALES data, not assimilation results. The three cases also have different incoming winds. They illustrate different flows, not the effect of changing geometry alone. Colors use fixed scales across cases. In the 2D views, dark outlined areas mark solid buildings, cyan lines locate the vertical section, and A–C mark virtual LES samples at exactly the same horizontal cells in both maps. The right-hand traces show horizontal speed sampled from those grid cells; they are not measured observations. In 3D, buildings use the matching STL geometry with no vertical exaggeration.</p>

<script>
if(new URLSearchParams(location.search).has('embedded')||document.documentElement.dataset.embedded)
  document.documentElement.classList.add('embedded');


const q=s=>document.querySelector(s),
  cases={
    train:{id:'US-Was-U1_d90',domain:'512 × 512 m',height:16},
    val:{id:'UA-KY-V7_d45',domain:'896 × 576 m',height:50},
    test:{id:'CH-BAS-V1_d00',domain:'896 × 768 m',height:90}
  };
let mode='3d';


function update(){
  const c=q('#case').value,d=cases[c],v=q('#film'),playing=!v.paused,t=v.currentTime||0;
  const path=(mode==='3d'?'flow3d_':'slices_')+c+'.mp4';
  v.pause();
  v.poster=(mode==='3d'?'poster3d_':'poster_')+c+'.jpg';
  v.src=path;
  v.onloadedmetadata=()=>{
    v.currentTime=Math.min(t,v.duration-.01);
    v.playbackRate=+q('#rate').value;
    if(playing)v.play().catch(()=>{});
    window.LESCharts?.draw()
  };
  q('#download').href=path;
  document.querySelectorAll('[data-view]').forEach(b=>{
    const on=b.dataset.view===mode;
    b.classList.toggle('active',on);
    b.setAttribute('aria-pressed',on)
  });
  q('.cinema').classList.toggle('slices-active',mode==='slices');
  q('#sensor-panels').hidden=mode!=='slices';
  window.LESCharts?.setCase(c);
  q('#geometry').innerHTML=`<strong>${[d.id](http://d.id/)}</strong><br>${d.domain} domain · buildings up to ${d.height} m`;
  q('#reading').innerHTML=mode==='3d'
    ?'<strong>Follow the flow around and above buildings.</strong><br>Streamlines reveal the shape of the instantaneous 3D flow. The near-ground slice shows speed at z = 2 m.'
    :'<strong>Look for wakes, shelter and vertical exchange.</strong><br>Compare horizontal speed at 2 m and 26 m. Below, warm colors show rising air and cool colors show sinking air. A–C are virtual LES samples, not observations.';
  q('#technique').textContent=mode==='3d'
    ?'Instantaneous streamlines show flow shape, not tracked particle paths. 120 saved velocity fields; approximately 30× real time. Speed includes u, v and w.'
    :'240 frames from linear interpolation between saved velocity components. Approximately 30× real time. Horizontal speed uses u and v; section colors show w. Sensor traces use the 120 actual saved fields.';
}


q('#case').onchange=update;
document.querySelectorAll('[data-view]').forEach(b=>b.onclick=()=>{mode=b.dataset.view;update()});
q('#rate').onchange=()=>q('#film').playbackRate=+q('#rate').value;
update();


const film=q('#film');
film.controls=false;
q('#play').onclick=()=>film.paused?film.play():film.pause();
q('#restart').onclick=()=>{film.currentTime=0;window.LESCharts?.draw()};
q('#scrub').oninput=e=>{film.currentTime=+e.target.value;window.LESCharts?.draw()};
film.onplay=()=>q('#play').textContent='Pause';
film.onpause=()=>q('#play').textContent='Play';
film.ontimeupdate=()=>{
  q('#scrub').value=film.currentTime;
  q('#clock').textContent='0:'+String(Math.floor(film.currentTime)).padStart(2,'0')+' / 0:20';
  window.LESCharts?.draw()
};
q('#full').onclick=()=>document.fullscreenElement
  ?document.exitFullscreen()
  :q('.cinema').requestFullscreen();


addEventListener('message',e=>{
  if(e.source===parent&&e.data?.type==='lecture:leave')film.pause()
});
addEventListener('keydown',e=>{
  if(parent!==window&&!['INPUT','SELECT'].includes(e.target.tagName)&&['ArrowLeft','ArrowRight','PageUp','PageDown'].includes(e.key)){
    e.preventDefault();
    parent.postMessage({type:'lecture:navigate',direction:['ArrowRight','PageDown'].includes(e.key)?1:-1},'*')
  }
});
</script>
